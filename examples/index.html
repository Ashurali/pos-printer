Install

npm install ble-pos-printer

ESM Import

import PrinterManager from 'ble-pos-printer';

const printer = new PrinterManager({
  storeName: 'MY STORE',
  paperWidth: 80,
  commandMode: 'auto',
});

await printer.connect();
await printer.printTest();

Detection API

// Get available tests
const tests = PrinterManager.getDetectionTests();

// Run a single test
await printer.runDetectionTest('escpos_full');

// Save result (persists per device)
printer.saveDetectionResult('escpos_full', {
  paperWidth: 80,
  autoCut: true,
});

Backend Sync (Any Framework)

const printer = new PrinterManager({
  apiEndpoint: '/api/printer_profiles',
  apiHeaders: {
    'X-CSRF-Token': document.querySelector(
      'meta[name=csrf-token]'
    ).content,
  },
});

// REST contract:
// GET    /api/printer_profiles       → list all
// PUT    /api/printer_profiles/:id   → upsert
// DELETE /api/printer_profiles/:id   → delete

Events

printer.on('connected', (d) => {
  console.log('Device:', d.device);
  console.log('Mode:', d.commandMode);
  console.log('Has profile:', d.hasProfile);
});

printer.on('printEnd', (d) => {
  console.log('Print OK:', d.type);
});

printer.on('error', (d) => {
  console.error(d.type, d.error);
});

React Hook

import { useEffect, useRef, useState } from 'react';
import PrinterManager from 'ble-pos-printer';

export function usePrinter(opts = {}) {
  const pm = useRef(null);
  const [connected, setConnected] = useState(false);

  useEffect(() => {
    pm.current = new PrinterManager(opts);
    pm.current.on('connected', () => setConnected(true));
    pm.current.on('disconnected', () => setConnected(false));
    return () => pm.current?.disconnect();
  }, []);

  return { printer: pm.current, connected };
}
